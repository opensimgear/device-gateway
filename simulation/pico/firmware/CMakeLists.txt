cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
  get_filename_component(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../../../.e2e/deps/pico-sdk" ABSOLUTE)
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  message(FATAL_ERROR "Pico SDK not found. Set PICO_SDK_PATH or initialize SDK 2.3.1 in device-gateway/.e2e/deps/pico-sdk.")
endif()
set(PICO_BOARD pico_w CACHE STRING "Pico board")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "UF2 is generated by build.mjs from the SDK's binary output")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(osg_pico_gateway C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
message(STATUS "Pico SDK ${PICO_SDK_VERSION_STRING}; tested SDK pin: 2.3.1 (079c6f39023649b154152db30f1d781e884879bc)")
add_executable(osg-pico-gateway src/main.c src/gateway.c src/usb_layout.c)
target_include_directories(osg-pico-gateway PRIVATE src)
target_compile_definitions(osg-pico-gateway PRIVATE PICO_STACK_SIZE=4096)
target_compile_options(osg-pico-gateway PRIVATE -Wall -Wextra -fstack-usage)
target_link_libraries(osg-pico-gateway PRIVATE pico_stdlib pico_unique_id tinyusb_device tinyusb_board)
pico_set_program_name(osg-pico-gateway "OpenSimGear Pico Gateway")
pico_set_program_version(osg-pico-gateway "0.1.0")
pico_enable_stdio_usb(osg-pico-gateway 0)
pico_enable_stdio_uart(osg-pico-gateway 0)
pico_add_extra_outputs(osg-pico-gateway)
