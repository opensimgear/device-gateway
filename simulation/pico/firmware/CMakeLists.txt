cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
  get_filename_component(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../../../.research/deps/pico-sdk" ABSOLUTE)
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  message(FATAL_ERROR "Pico SDK not found. Run build.mjs or set PICO_SDK_PATH.")
endif()
set(PICO_BOARD pico_w CACHE STRING "Pico board")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "UF2 is generated by build.mjs from the SDK's binary output")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(osg_pico_gateway C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
message(STATUS "Pico SDK ${PICO_SDK_VERSION_STRING}; tested SDK pin: 2.3.1 (079c6f39023649b154152db30f1d781e884879bc)")
get_filename_component(OSG_GATEWAY_FIRMWARE "${CMAKE_CURRENT_LIST_DIR}/../../../firmware" ABSOLUTE)
if(NOT OSG_COMMON_SOURCE OR NOT OSG_DEVICE_SOURCE)
  message(FATAL_ERROR "Set OSG_COMMON_SOURCE and OSG_DEVICE_SOURCE, or use build.mjs to provision pinned sources.")
endif()
add_executable(osg-pico-gateway
  src/main.c src/debug_usb.c src/pico_storage.c src/debug_gateway.c src/virtual_device.c
  ${OSG_GATEWAY_FIRMWARE}/src/gateway.c
  ${OSG_GATEWAY_FIRMWARE}/src/gateway_registry.c
  ${OSG_GATEWAY_FIRMWARE}/src/gateway_layout.c
  ${OSG_GATEWAY_FIRMWARE}/src/gateway_cockpit.c
  ${OSG_GATEWAY_FIRMWARE}/src/gateway_journal.c
  ${OSG_GATEWAY_FIRMWARE}/shared-cyphal/src/gateway_protocol.c
  ${OSG_DEVICE_SOURCE}/firmware/src/runtime_config.c
  ${OSG_DEVICE_SOURCE}/firmware/src/device_actions.c
  ${OSG_COMMON_SOURCE}/src/inputs/profile.c
  ${OSG_COMMON_SOURCE}/src/inputs/inputs.c
  ${OSG_COMMON_SOURCE}/src/inputs/button.c
  ${OSG_COMMON_SOURCE}/src/events/events.c
  ${OSG_COMMON_SOURCE}/src/actions/actions.c
  ${OSG_COMMON_SOURCE}/src/usb/descriptor.c)
target_include_directories(osg-pico-gateway PRIVATE src
  ${OSG_GATEWAY_FIRMWARE}/include
  ${OSG_GATEWAY_FIRMWARE}/shared-cyphal/include
  ${OSG_GATEWAY_FIRMWARE}/shared-cyphal/generated
  ${OSG_DEVICE_SOURCE}/firmware/include
  ${OSG_COMMON_SOURCE}/include)
target_compile_definitions(osg-pico-gateway PRIVATE PICO_STACK_SIZE=24576
  PICO_FLASH_ASSUME_CORE1_SAFE=1)
# The SDK's default core-0 stack lives in a 4 KiB scratch bank. Reserve a
# contiguous 24 KiB stack ending at physical SRAM top, with no second core.
pico_set_linker_script_var(osg-pico-gateway RAM_LENGTH 0x3c000)
pico_set_linker_script_var(osg-pico-gateway SCRATCH_X_ORIGIN 0x2003c000)
pico_set_linker_script_var(osg-pico-gateway SCRATCH_X_LENGTH 0)
pico_set_linker_script_var(osg-pico-gateway SCRATCH_Y_ORIGIN 0x2003c000)
pico_set_linker_script_var(osg-pico-gateway SCRATCH_Y_LENGTH 0x6000)
target_compile_options(osg-pico-gateway PRIVATE -Wall -Wextra -fstack-usage)
target_link_options(osg-pico-gateway PRIVATE "LINKER:-T,${CMAKE_CURRENT_LIST_DIR}/storage.ld")
target_link_libraries(osg-pico-gateway PRIVATE pico_stdlib pico_unique_id pico_flash hardware_flash tinyusb_device tinyusb_board)
pico_set_program_name(osg-pico-gateway "OpenSimGear Pico Debug Gateway")
pico_set_program_version(osg-pico-gateway "0.1.0")
pico_enable_stdio_usb(osg-pico-gateway 0)
pico_enable_stdio_uart(osg-pico-gateway 0)
pico_add_extra_outputs(osg-pico-gateway)
